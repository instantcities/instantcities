{%- comment -%} Links every landmark of a city and every famous piece of art in it that a text names, once each: {% include link-landmarks.html text=t city='paris' landmarks=list %} prints the text, list being _data/landmarks/.json (see link-first.html for what counts as a mention). The art is the pieces of _data/art.json in the city's museums. {%- endcomment -%} {%- assign ll_text = include.text -%} {%- for ll in include.landmarks -%} {%- assign ll_url = '/' | append: include.city | append: '/landmark/' | append: ll.id | append: '/' -%} {%- capture ll_text %}{% include link-first.html text=ll_text name=ll.name url=ll_url %}{% endcapture -%} {%- endfor -%} {%- assign ll_museums = site.data.art.museums | where: 'city', include.city -%} {%- for ll_m in ll_museums -%} {%- assign ll_pieces = site.data.art.pieces | where: 'museum', ll_m.id -%} {%- for ll_a in ll_pieces -%} {%- assign ll_url = '/art/' | append: ll_a.id | append: '/' -%} {%- assign ll_name = ll_a.title | remove_first: 'The ' -%} {%- capture ll_text %}{% include link-first.html text=ll_text name=ll_name url=ll_url %}{% endcapture -%} {%- endfor -%} {%- endfor -%} {{- ll_text -}}
